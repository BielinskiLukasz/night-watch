    Implement computeTifBoundsHistory(dayRecords, settings, activityLog):
      Sort by date string (lexicographic — same as accuracy.js).
      minDays = settings.tifRollingDays ?? settings.minDays.
      Loop for (let i = minDays; i < sorted.length; i++):
        history = sorted.slice(0, i)  // look-ahead prevention
        actual = sorted[i]
        pred = tifForecast(history, settings, activityLog ?? {})
        if (pred.isColdStart): push { date:actual.date, wake:null, napStart:null, napEnd:null, bedtime:null }; continue
        Build entry = { date: actual.date }
        For each type in EVENT_TYPES:
          p = pred[type]
          entry[type] = (p && p.algMin != null && p.algMax != null)
            ? { algMin: p.algMin, algMax: p.algMax, central: p.central ?? null, precisionScore: p.precisionScore ?? null }
            : null
        results.push(entry)
      Return results.

    Implement computeTifAccuracy(history, dayRecords):
      Build Map<date, dayRecord> from dayRecords for O(1) actual lookup.
      Initialize counters for each EVENT_TYPE: { total:0, windowHitCount:0, widthSum:0, highConfCount:0 }.
      For each entry in history:
        For each type in EVENT_TYPES:
          bounds = entry[type]; if null, skip (not counted).
          actualDay = dayByDate.get(entry.date); if absent, skip.
          actualSlot = actualDay[type]; if null, skip.
          actualMinutes: extract HH:MM from slot (event object or bare string), then timeToMinutes.
          algMinMin = timeToMinutes(bounds.algMin); algMaxMin = timeToMinutes(bounds.algMax).
          counters[type].total++.
          if (actualMinutes >= algMinMin && actualMinutes <= algMaxMin): counters[type].windowHitCount++.
          counters[type].widthSum += algMaxMin - algMinMin.
          if (bounds.precisionScore != null && bounds.precisionScore >= 80): counters[type].highConfCount++.
      Build result: for each type:
        t = counters[type].total
        pct(n) = t===0 ? 0 : Math.round(n/t*100)
        result[type] = { windowHit:{count:c.windowHitCount, pct:pct(c.windowHitCount)}, avgWidthMin: t===0 ? 0 : c.widthSum/t, highConf:{count:c.highConfCount, pct:pct(c.highConfCount)} }
      Return result.
