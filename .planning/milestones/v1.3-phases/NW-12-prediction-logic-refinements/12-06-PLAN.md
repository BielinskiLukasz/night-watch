    Signal definitions (D-13):

    Signal 1 — napFrequency (weight 0.40):
      input: pre-filtered dayRecords (stage-scoped, caller applies filterDayRecordsByStage before calling napProbability)
      formula: days with napStart != null / total days; range 0.0–1.0

    Signal 2 — elapsedWakeTime (weight 0.30):
      input: context.todayWakeHHMM (today's wake time as HH:MM or null), context.currentHour, context.currentMinute
      napStart percentiles from the dayRecords window (calculatePercentiles on napStart times)
      formula: if todayWakeHHMM null or napStart history null → signal = 0
        wakeMins = timeToMinutes(todayWakeHHMM)
        p10_ns = napStart P10 in minutes; p90_ns = napStart P90 in minutes
        typical MA window = [p10_ns - wakeMins, p90_ns - wakeMins] (minutes of activity before nap)
        nowMins = currentHour * 60 + currentMinute
        elapsed = nowMins - wakeMins (how long awake since wake)
        signal = clamp(elapsed / (p90_ns - p10_ns), 0, 1) when p90_ns > p10_ns; else 0

    Signal 3 — noNapStreak penalty (weight 0.20):
      input: context.napStreak (consecutive days ending yesterday without a nap, integer >= 0)
      formula: max(0, 1 - napStreak / 5); clamped to [0, 1]
      → streak=0 → 1.0; streak=5+ → 0.0

    Signal 4 — windowPassed (weight 0.10):
      input: context.currentHour, context.currentMinute, napStart P90 in minutes
      formula: if currentTime > p90_ns → signal = 0 (window closed); else signal = 1
      → entire function returns 0 when this signal fires AND collapses total to 0

    Total (before window-passed collapse):
      raw = signal1 * 0.40 + signal2 * 0.30 + signal3 * 0.20 + signal4 * 0.10
      score = Math.round(raw * 100)  // single round at the end

    Window-passed collapse:
      if nowMins > p90_ns → return 0  (represents "nap window closed")
      NOTE: returning 0 (integer) means today-screen can distinguish "0 = window closed" from
      null = "cannot compute". Use 0 for window-closed and null for cold-start/no-data.

    Test cases:
    - All days have nap: freq=1.0, streak=0, elapsed in window, window open → score near 73
    - No days have nap: freq=0.0 → low score regardless of other signals
    - window already passed: → returns 0
    - napStreak=5: signal3=0 → weight 0.20 zeroed
    - todayWakeHHMM=null: signal2=0 → weight 0.30 zeroed
    - dayRecords empty: → return null

    ```javascript
    /**
     * Compute a nap probability score (0–100 integer) for today.
     *
     * PRED-12 / D-13: additive 4-signal scoring.
     *
     * @param {object[]} dayRecords  stage-scoped day records (caller applies filterDayRecordsByStage)
     * @param {object}   settings    settings snapshot (uses minDays, windowDays)
     * @param {object}   context     today's state
     * @param {string|null} context.todayWakeHHMM  today's wake time as 'HH:MM' or null
     * @param {number}   context.currentHour    current hour 0-23
     * @param {number}   context.currentMinute  current minute 0-59
     * @param {number}   [context.napStreak=0]  consecutive no-nap days ending yesterday
     * @returns {number|null} 0–100 integer, or null if score cannot be computed
     */
    export function napProbability(dayRecords, settings, context) {
      if (!dayRecords || dayRecords.length === 0) return null;

      const { windowDays, minDays } = settings;
      const window = windowDays != null && dayRecords.length > windowDays
        ? dayRecords.slice(dayRecords.length - windowDays)
        : dayRecords;

      if (window.length < minDays) return null;

      const {
        todayWakeHHMM = null,
        currentHour = 0,
        currentMinute = 0,
        napStreak = 0,
      } = context || {};

      const nowMins = currentHour * 60 + currentMinute;

      // ── Signal 1: nap frequency (40%) ────────────────────────────────────
      const totalDays = window.length;
      const napDays = window.filter(d => extractTime(d.napStart) != null).length;
      const sig1 = totalDays > 0 ? napDays / totalDays : 0;

      // ── napStart percentiles (needed by sig2 and sig4) ───────────────────
      const nsResult = calculatePercentiles(window, d => extractTime(d.napStart));
      const p10_ns = nsResult ? nsResult.min : null;
      const p90_ns = nsResult ? nsResult.max : null;

      // ── Signal 4: window-passed (10%) — check early for early return ─────
      const windowOpen = p90_ns === null || nowMins <= p90_ns;
      const sig4 = windowOpen ? 1 : 0;
      // D-13: window-passed collapses entire score to 0
      if (!windowOpen) return 0;

      // ── Signal 2: elapsed wake time position (30%) ───────────────────────
      let sig2 = 0;
      if (todayWakeHHMM && p10_ns !== null && p90_ns !== null && p90_ns > p10_ns) {
        const wakeMins = timeToMinutes(todayWakeHHMM);
        const elapsed = nowMins - wakeMins;
        const windowWidth = p90_ns - p10_ns;
        sig2 = Math.max(0, Math.min(1, elapsed / windowWidth));
      }

      // ── Signal 3: no-nap streak penalty (20%) ────────────────────────────
      const sig3 = Math.max(0, 1 - napStreak / 5);

      // ── Combine and round once ────────────────────────────────────────────
      const raw =
        sig1 * NAP_SCORE_WEIGHTS.napFrequency +
        sig2 * NAP_SCORE_WEIGHTS.elapsedWakeTime +
        sig3 * NAP_SCORE_WEIGHTS.noNapStreak +
        sig4 * NAP_SCORE_WEIGHTS.windowPassed;

      return Math.round(raw * 100);
    }
    ```
