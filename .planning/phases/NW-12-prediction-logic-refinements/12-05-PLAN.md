    context = {isIntenseToday: false}
    → no modifier; normal full-window bedtime percentiles used

    context = {isIntenseToday: true}, 10 intense-day records in window (>= minDays)
    → bedtime.central = P50 of those 10 intense-day bedtimes (not shifted by offset)

    context = {napStartLogged: false, currentHour: 19, isIntenseToday: false}
    settings.eveningHour = 18
    → no-nap branch fires (19 >= 18 and no nap logged)
    → filter window to days where napStart is null; compute P50 bedtime
    → if no-nap sub-window < minDays → fallback: full-window P50 - noNapBedtimeOffsetMinutes(30)

    context = {napStartLogged: true, currentHour: 19}
    → PRED-11 does NOT fire (nap was logged today)

    context = {napStartLogged: false, currentHour: 17}
    settings.eveningHour = 18
    → PRED-11 does NOT fire (evening not reached)

    Precedence: when both isIntenseToday=true AND PRED-11 fires
    → PRED-11 (no-nap shift) takes precedence — no-nap is a stronger contextual signal
    → isIntenseToday modifier does not additionally apply when no-nap shift is active

    ```javascript
    /**
     * Compute a bedtime band from a filtered sub-window of day records.
     * If sub-window has fewer than minDays records, shifts the full-window P50 bedtime
     * by -fallbackOffsetMinutes and applies the same offset to min/max.
     * Returns {central, min, max} as minute values, or null if full window has no bedtime data.
     * D-03 / D-08 pattern.
     */
    function subWindowBedtime(window, filterFn, fallbackOffsetMinutes, settings) {
      const { minDays } = settings;
      const subWin = window.filter(filterFn);
      if (subWin.length >= minDays) {
        // Enough intense/no-nap days — use their own percentiles
        return calculatePercentiles(subWin, d => extractTime(d.bedtime));
      }
      // Thin history — shift full-window percentiles by fixed offset
      const base = calculatePercentiles(window, d => extractTime(d.bedtime));
      if (base === null) return null;
      return {
        central: base.central - fallbackOffsetMinutes,
        min:     base.min    - fallbackOffsetMinutes,
        max:     base.max    - fallbackOffsetMinutes,
      };
    }
    ```
