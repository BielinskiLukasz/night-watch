function renderCell(value, colDef, snap, isMinMaxDate = false) {
  const td = document.createElement('td');
  if (colDef.isTime && value) {
    td.textContent = formatTime(value, snap.timeFormat);
  } else if (colDef.isRatio && value != null) {
    td.textContent = value.toFixed(2);
  } else if (!colDef.isTime && !colDef.isRatio && value != null) {

}
