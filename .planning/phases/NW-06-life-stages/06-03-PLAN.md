    3. Add a renderStageSelector function (module-private, before mountTodayScreen or as a closure inside it) with this logic:
    
    ```js
    function renderStageSelector(container, stages, activeStageId) {
      if (!container) return;
      if (!stages || stages.length === 0) {
        container.style.display = 'none';
        return;
      }
      container.style.display = '';
      
      // Build/update the select — clear first, rebuild  
      clear(container);
      
      const wrapper = el('div', { className: 'stage-selector-wrapper' });
      
      const label = el('label', { className: 'stage-selector-label', textContent: 'Showing data for: ' });
      const select = el('select', { className: 'stage-select', 'aria-label': 'Select data stage' });
      
      // "All data" option (D6-12)
      const allOpt = el('option', { value: '', textContent: 'All data' });
      if (!activeStageId) allOpt.selected = true;
      select.appendChild(allOpt);
      
      for (const stage of stages) {
        const opt = el('option', { value: stage.id, textContent: stage.name }); // textContent — never innerHTML
        if (stage.id === activeStageId) opt.selected = true;
        select.appendChild(opt);
      }
      
      label.appendChild(select);
      wrapper.appendChild(label);
      container.appendChild(wrapper);
      
      // Fallback note placeholder (shown/hidden in render())
      const note = el('p', { id: 'stage-fallback-note', className: 'stage-fallback-note' });
      note.style.display = 'none';
      container.appendChild(note);
      
      // onChange: persist selection and trigger re-render via settings.subscribe
      select.addEventListener('change', () => {
        const newId = select.value || null; // empty string → null = "All data"
        settings.update({ activeStageId: newId });
      });
    }
    ```

    4. In the render() function inside mountTodayScreen(), update the forecast computation block:
    
    Before:
    ```js
    const forecastDays = eventLog.daysBySubjectiveNight(snap.cutoverHour);
    const predictions = forecast(forecastDays, snap);
    renderForecastSection(predictions, snap, forecastDays, nextEventCard, coldStartMsg, forecastCards);
    ```
    
    After:
    ```js
    const allForecastDays = eventLog.daysBySubjectiveNight(snap.cutoverHour);
    
    // D6-11: stage filter + thin-stage fallback
    const stages = snap.stages || [];
    const activeStageId = snap.activeStageId || null;
    let forecastDays = allForecastDays;
    let thinStage = false;
    
    if (activeStageId) {
      const filtered = filterDayRecordsByStage(allForecastDays, stages, activeStageId);
      const validCount = filtered.filter(d => !d.rejected).length;
      if (validCount < (snap.minDays || 7)) {
        thinStage = true;
        // Fall back to all data (D6-11)
        forecastDays = allForecastDays;
      } else {
        forecastDays = filtered;
      }
    }
    
    // Render/update stage selector
    renderStageSelector(stageSelectorContainer, stages, activeStageId);
    
    // Update fallback note visibility
    const noteEl = document.getElementById('stage-fallback-note');
    if (noteEl) {
      if (thinStage) {
        noteEl.textContent = 'Not enough data in this stage — showing all data.'; // textContent only
        noteEl.style.display = '';
      } else {
        noteEl.style.display = 'none';
      }
    }
    
    const predictions = forecast(forecastDays, snap);
    renderForecastSection(predictions, snap, forecastDays, nextEventCard, coldStartMsg, forecastCards);
    ```

    NOTE: renderStageSelector clears and rebuilds the container on every render(). This is intentional — since render() is called by the subscriber on every settings or eventLog change, rebuilding the selector ensures the selected option always reflects the current snap.activeStageId. The change handler calls settings.update(), which triggers the subscriber, which calls render() — this is the same reactive loop already used by the grouping toggle.
