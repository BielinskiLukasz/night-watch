    renderColdStart(root, remaining): root.replaceChildren(); const p = document.createElement('p'); p.className = 'coldStartNote'; p.textContent = 'Not enough history to compute accuracy — keep logging! (' + remaining + ' more day(s) needed)'; root.appendChild(p);

    renderStageBadge(badge, snap): if (snap.activeStageId) { const stage = (snap.stages||[]).find(s=>s.id===snap.activeStageId); badge.hidden = !stage; if(stage) badge.textContent = 'Viewing: ' + stage.name; } else { badge.hidden = true; }

    buildAccuracyGrid(gridEl, result, snap): gridEl.replaceChildren().

    Column header row: first cell = empty th. Then for each col in ACCURACY_COLS: cell = document.createElement('div'); cell.className = 'accHeader'; cell.textContent = col.header; gridEl.appendChild(cell).

    For each row in ACCURACY_ROWS: row label cell = document.createElement('div'); cell.className = 'accRowLabel'; cell.textContent = row.label; gridEl.appendChild(cell). rowResult = result[row.type]. isNapType = NAP_TYPES.has(row.type). showDash = isNapType && rowResult.total < snap.minDays. For each col in ACCURACY_COLS: cell = document.createElement('div'); cell.className = 'accCell'. if (showDash) { cell.textContent = '—'; } else if (rowResult.total === 0) { cell.textContent = '—'; } else { pctEl = document.createElement('span'); pctEl.className = 'accPct'; pctEl.textContent = rowResult[col.key].pct + '%'; countEl = document.createElement('small'); countEl.className = 'accCount'; countEl.textContent = 'n=' + rowResult.total; cell.append(pctEl, document.createElement('br'), countEl); } gridEl.appendChild(cell).

    mountAccuracyScreen({ root, eventLog, settings }): root.replaceChildren(). Create: stageBadge = document.createElement('p'); stageBadge.className = 'stageChip'; stageBadge.hidden = true. gridRoot = document.createElement('div'); gridRoot.className = 'accuracyGrid'. root.replaceChildren(stageBadge, gridRoot).

    render(): snap = settings.get(). allDays = eventLog.daysBySubjectiveNight(snap.cutoverHour). days = snap.activeStageId ? filterDayRecordsByStage(allDays, snap.stages||[], snap.activeStageId) : allDays. validCount = days.filter(d=>!d.rejected).length. if (validCount < snap.minDays) { renderColdStart(root, snap.minDays - validCount); return; } renderStageBadge(stageBadge, snap). const result = computeAccuracy(days, snap). buildAccuracyGrid(gridRoot, result, snap).

    Wire subscriptions: render(); const unsubLog = eventLog.subscribe(render); const unsubSettings = settings.subscribe(render); return { unsubscribe() { unsubLog(); unsubSettings(); } };

    ALL cell content via textContent. No innerHTML. No SVG (accuracy grid is pure HTML per D7-14 — it is a "compact stats table, not a chart").

    filterDayRecordsByStage THREE ARGS: filterDayRecordsByStage(allDays, snap.stages||[], snap.activeStageId). Never two args. (RESEARCH Pitfall 1.)
